    client = p.connect(p.GUI)
    p.setGravity(0, 0, -9.8, physicsClientId=client)
    humanoid = Humanoid(client)
    humanoid.reset([0, 0, 1.5])
    humanoid.apply_action([0.1] * 10)

    joint_sliders = {}
    for i, idx in enumerate(humanoid.joint_index):
        joint_name = p.getJointInfo(humanoid.id, idx)[1].decode()
        lo = p.getJointInfo(humanoid.id, idx)[8]
        hi = p.getJointInfo(humanoid.id, idx)[9]
        if hi > lo:  # only create sliders with valid limits
            joint_sliders[joint_name] = p.addUserDebugParameter(joint_name, lo, hi, 0.0)
        else:
            joint_sliders[joint_name] = p.addUserDebugParameter(joint_name, -3.14, 3.14, 0.0)

    p.resetBasePositionAndOrientation(humanoid.id, [0, 0, 4], p.getQuaternionFromEuler([1.57, 0, 0]))
    for _ in range(240):
        p.stepSimulation()
        time.sleep(1. / 240.)

    while p.isConnected():
        target_angles = []
        for joint_name in joint_sliders:
            val = p.readUserDebugParameter(joint_sliders[joint_name])
            target_angles.append(val)

        humanoid.apply_position(target_angles)
        p.stepSimulation()
        obs = humanoid.get_obs()
        print("Observation:", obs)
        time.sleep(1. / 240.)
